-- 1. Todas las tareas del departamento
SELECT
    te.id,
    CONCAT(e.nombres, ' ', e.apellidos) AS empleado,
    c.nombre AS cargo,
    te.titulo, te.estado, te.prioridad,
    te.fecha_creacion, te.fecha_pactada, te.fecha_entregada,
    te.aprobada, te.rechazada, te.pausada, te.reprogramada,
    te.observacion_entrega
FROM tareas_empleados te
JOIN empleados e ON te.empleado = e.id
LEFT JOIN cargos c ON e.cargo = c.id
WHERE e.departamento = 1
ORDER BY te.estado DESC, te.fecha_creacion DESC;

-- 2. Completadas en abril y mayo 2026
SELECT te.id, CONCAT(e.nombres,' ',e.apellidos) AS empleado,
    te.titulo, te.fecha_pactada, te.fecha_entregada,
    DATEDIFF(te.fecha_entregada, te.fecha_pactada) AS dias_diferencia
FROM tareas_empleados te
JOIN empleados e ON te.empleado = e.id
WHERE e.departamento = 1
  AND te.estado = 'Completada'
  AND YEAR(te.fecha_entregada) = 2026
  AND MONTH(te.fecha_entregada) IN (4, 5)
ORDER BY te.fecha_entregada;

-- 3. Tareas en proceso con días restantes
SELECT te.id, CONCAT(e.nombres,' ',e.apellidos) AS empleado,
    te.titulo, te.fecha_pactada,
    DATEDIFF(te.fecha_pactada, CURDATE()) AS dias_restantes
FROM tareas_empleados te
JOIN empleados e ON te.empleado = e.id
WHERE e.departamento = 1
  AND te.estado = 'En Proceso'
ORDER BY dias_restantes;

-- 4. Producción mensual por fecha de entrega
SELECT
    YEAR(te.fecha_entregada) AS anio,
    MONTH(te.fecha_entregada) AS mes,
    COUNT(*) AS total_entregas
FROM tareas_empleados te
JOIN empleados e ON te.empleado = e.id
WHERE e.departamento = 1
  AND te.estado = 'Completada'
  AND te.fecha_entregada IS NOT NULL
GROUP BY YEAR(te.fecha_entregada), MONTH(te.fecha_entregada)
ORDER BY anio, mes;
